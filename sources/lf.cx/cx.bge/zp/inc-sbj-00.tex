\begin{frame}  
  \frametitle{LFCX:BGE:Manifest für die GS-LDK}

  \begin{itemize}
    \item Wir halten Gesetze und Verordnungen ein:
      \begin{itemize}
        \footnotesize
        \item Schulpflicht, keine Gewalt, kein Mobbing, keine Störungen, ...
        \item Klausurteilnahme
        \item Entschuldigungen für Fehltage
        \item Beachtung der Höchstgrenze für Fehltage
        \item ...
      \end{itemize}
    \item Jede Schülerin erhält im Zeugnis bedingungslos eine 3.
    \item Klausurenergebnisse werden normiert auf die Noten 3 - 1.
    \item Für die mündliche Note sammeln die Schülerinnen Punkte:
      \begin{itemize}
        \footnotesize
      \item eine 3 : keine Punkte
      \item eine 2 : so viele Punkte, wie Unterrichtstage 
      \item eine 1 : 2,5 mal so viele Punkte, wie Unterrichtstage 
    \end{itemize}
    \end{itemize}

\end{frame}



\begin{frame}  
  \frametitle{LFCX:BGE:Möglichkeiten zum Punktesammln}

  \begin{itemize}
    \item Sehr aktive mündliche Mitarbeit: 1P/pro Schultag.
    \item Große Präsentation: 
      \begin{itemize}
      \footnotesize
      \item Unterlagen 1 Tag vorab: 1P
      \item Präsentieren: 1P
      \item inhaltliche Qualität, Nützlichkeit für Mitschüler, Präsentationsreife: je 1P
      \end{itemize}
    \item Große Hausarbeit: 1 - 3 Punkte nach Qualität
    \item Übungen (Diagramme etc.), Zusammenfassungen: 1P
    \item Supportive Behavior, besondere Unterstützung von Klassenkameradinnen: 2P
    \item Was mir ein unmittelbares *Wow* abnötigt: 1P
  \end{itemize}

\end{frame}
