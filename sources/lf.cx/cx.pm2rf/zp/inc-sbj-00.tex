\begin{frame}
  \frametitle{Etwas Projektmanagement: Definition! }
	
	\begin{block}{Ein \textbf{Projekt} ist}
	\vspace{0.5cm}

		\begin{itemize}
			\item ein irgendwie Vorhaben
			\item mit einem dedizierten Ende 
			\item  und einem einzigartigen Ziel,
			\item (zu dessen Erreichen Resourcen verbraucht werden.)
		\end{itemize}
	\end{block}
\end{frame}

\begin{frame}
  \frametitle{Etwas Projektmanagement: Abgrenzung}
	
	\begin{block}{\textbf{Kein} Projekt ist}
	\vspace{0.5cm}
		
		\begin{itemize}
			\item ein 'Irgend-wann-mal'-Vorhaben (ohne Ende)
			\item eine wiederholbare (Linien)Tätigikeit (ohne einzigartiges Ziel) bzw.
			\item ein Prozess mit 'immer' gleichem Ablauf
		\end{itemize}
	\end{block}
\end{frame}

\begin{frame}
  \frametitle{Etwas Projektmanagement: Struktur}
	
	\begin{block}{Ein Projekt braucht}
	\vspace{0.5cm}
		
		\begin{itemize}
			\item eine\footnote{Männer sind mitgemeint}, die das Ziel vorgibt (Projekteignerin)
			\item eine, die die Resourcen bereitstellt (Projektsponsorin)
			\item eine, die den (ja unbekannten) Ablauf organisiert (Projektmanagerin)
		\end{itemize}
	\end{block}
\end{frame}

\begin{frame}
  \frametitle{Etwas Projektmanagement: Tradition}
	
	In Deutschland gibt es zwei 'Projekt-Management-Communities':

	\begin{block}{GPM = Deutsche Gesellschaft für Projektmanagement}
	\vspace{0.5cm}
		
		\begin{itemize}
			\item sehr auf das 'Wie' und 'Definitionen' ausgerichtet
			\item kommt vom 'Wasserfall'-Modell her
			\item eher deutsch ausgerichtet
			\item \href{https://www.gpm-ipma.de/}{https://www.gpm-ipma.de/}
		\end{itemize}
	\end{block}

	\begin{block}{PMI = Project Management Institut}
	\vspace{0.5cm}
		
		\begin{itemize}
			\item sehr auf Prozess- und Begriffsdefinitionen konzentriert
			\item kommt vom 'Wasserfall'-Modell her
			\item internatiale Organisation, bei der man sich als PMP zertifiziert
			\item \href{https://www.pmi.org/}{https://www.pmi.org/}
		\end{itemize}
	\end{block}
\end{frame}

\begin{frame}
  \frametitle{Die WBS oder 'Wie geht das'}
	\begin{block}{Eine Work-Break-Down-Structure}
		\begin{itemize}
			\item Clustert alle zu erledigenden Aufgaben, indem sie die explizit einer Gruppe zuordnet, oder sie erst einmal unter eine Gruppe mitdenkt.
			\item Abstrahiert von zeitlichen und logischen Zusammenhängen
			\item Will das initiale 'Tätigkeitswirrwar' ordnen, in dem es Aufgaben \textbf{irgendwie} zu Gruppen zusammenführt.
			\item Nutzt die eingeführten Gruppen als Assoziationshilfen, um das zu finden, was auch noch getan werden muss.
		\end{itemize}
	\end{block}

	Einfaches Tools\footnote{vgl. auch \href{https://www.drawio.com/blog/project-planning-diagrams}{https://www.drawio.com/blog/project-planning-diagrams}}
	\begin{itemize}
		\item für die textuelle Erfassung: \textit{eingerückte unsortierte Listen} (\textbf{Markdown})
		\item für die graphische Darstellung: \textit{Baum} (\textbf{drawio})
		\item für beides: \textbf{Zettel und Papier}
	\end{itemize}
\end{frame}


\begin{frame}
  \frametitle{Etwas Projektmanagement: Revolution}
	
	Vorherrschendes agiles 'Projekt-Management-Modell':

	\begin{block}{Scrum bzw. Scrum.org /* kein Acronym */ }
	\vspace{0.5cm}
		
		\begin{itemize}
			\item löst Vorhaben in kleinen iterativen Feedbackschleifen
			\item will das 'Wasserfall'-Problemlösen
			\item entstanden aus dem Unwohlsein der Web-Sites-Entwickler mit dem traditionellen Projektmanagement
			\item ist letztlich eine Time-\&Material-Beauftragungsmodell
			\item bietet Zertifizierungen zum Scrum-Master, -Projectowner, ...
			\item \href{https://www.scrum.org/}{https://www.scrum.org/}
		\end{itemize}
	\end{block}
\end{frame}

\begin{frame}
  \frametitle{Etwas Projektmanagement: "Variante"}

	die oft (fälschlicherweise) als Alternative zu \textit{Scrum} gehandelte \href{https://de.wikipedia.org/wiki/Kanban}{Methode}
	\begin{block}{Kanban}
	\vspace{0.5cm}
		
		\begin{itemize}
			\item organisiert das agile Abarbeiten von Prozessen
			\item arbeitet mit Lanes und Karten, die weiter geschoben werden
			\item kommen aus dem japanischen Automobilbau
			\item gibt es als freie Opensource-Software, wie z.B. wekan \href{https://wekan.github.io/}{https://wekan.github.io/}
			\item ist keine Alternative für Scrum, sondern etwas ganz anderes
		\end{itemize}
	\end{block}
\end{frame}
